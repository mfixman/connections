\begin{frame}{Knights and Knaves}
	\begin{center}
		\includegraphics[height=100pt]{knights_knaves_small.jpg}
	\end{center}

	\vfill

	A traveller meets a husband and wife.
	He knows that either both of them are \textit{knights}, who always tell the truth, or \textit{knaves}, who always lie.

	The husband makes a statement, and the traveller is given three clues.
	\vspace{-5pt}
	\begin{enumerate}
		\item If the husband's statement is false, then his wife is a knight.
		\item If the wife is a knight, then the husband's statement is true.
		\item If the husband is a knight, then his wife is a knight
	\end{enumerate}
	\sourcecite{Smullyan78}
\end{frame}

\begin{frame}{Knights and Knaves}
	\begin{center}
		\Large\texttt{PUZ020-1.p}
	\end{center}

	{\large
	\begin{gather}
		a(\eta) \qquad \neg b(\eta) \notag\\
		\neg a(A) \lor \neg c\bigl(\phi(A)\bigr) \lor b(A) \notag\\
		c\bigl(\phi(\eta)\bigr) \lor b(\gamma)
		\qquad
		\neg b(\gamma) \lor c\bigl(\phi(\eta)\bigr) \notag
	\end{gather}
	}

	\begin{multicols}{2}
		\begin{description}
			\item[$a(A)$] $A$ is a person.
			\item[$b(A)$] $A$ is a knight.
			\item[$c(B)$] $B$ is a true statement.
			\item[$d(A)$] $A$ is a knave.
			\item[$e(A,B)$] $A$ says statement $B$.
			\item[$\phi(A)$] statement $\phi$ by $A$.
			\item[$\eta$] the husband.
			\item[$\gamma$] the wife.
		\end{description}
	\end{multicols}
	\sourcecite{Smullyan78,tptp}
\end{frame}

\begin{frame}{Further Improvements to SATCoP}
	\begin{itemize}
		\item \satcopz{} uses the SAT solver to check whether the ground clauses are inconsistent.
		\item \satcop{} also looks at the current SAT model.
		\item When choosing the next connection, it tries a goal that is true in that model. If the model gives no preference, it keeps the shuffled order.
	\end{itemize}
	\sourcecite{satcop}
\end{frame}

\section{SATResetCoP}
\begin{frame}{SATResetCoP}
	\begin{itemize}
		\item Keep newly generated ground clauses.
		\item \step{Reset} the tableau instead of locally backtracking.
		\item Increase the depth only when an attempt produces nothing new.
	\end{itemize}

	\vfill
	\begin{beamercolorbox}[rounded=true,sep=7pt,center]{block body}
		SAT refutations remain \textbf{sound}; the reset search is \textbf{incomplete}.
	\end{beamercolorbox}
\end{frame}

	{\large
	\begin{gather}
		\neg p(X) \lor \neg p(\psi(X)) \notag\\
		p(\psi(X)) \lor p(X) \notag\\
		\neg p(X) \lor p(\psi(X)) \notag
	\end{gather}
	}

	\begin{description}
		\item[$p(X)$] an arbitrary unary atom.
		\item[$\psi(X)$] an arbitrary unary function.
		\item[$X$] a variable, universally quantified in every clause.
		\item[$\kappa$] the fresh constant used to ground clauses for the SAT shadow.
	\end{description}
	\sourcecite{Pelletier86}

\section{Takeaways}
\begin{frame}{Takeaways}
	\centering
	{\LARGE\bfseries Keep the clauses, \step{reset} the tree.}

	\vfill
	\raggedright
	\begin{itemize}
		\item Aggressive resets help, but change which problems are solved.
		\item Schedule \step{Reset} search before conventional \satcop{} or \leancop{} search.
