\begin{frame}{Base Problem}
	\begin{center}
		\includegraphics[height=100pt]{knights_knaves_small.jpg}
	\end{center}

	\vfill

	A traveller meets a husband and wife.
	He knows that either both of them are \textit{knights}, who always tell the truth, or \textit{knaves}, who always lie.

	The husband makes a statement, and the traveller is given three clues.
	\vspace{-5pt}
	\begin{enumerate}
		\item If the husband's statement is false, then his wife is a knight.
		\item If the wife is a knight, then the husband's statement is true.
		\item If the husband is a knight, then his wife is a knight
	\end{enumerate}
\end{frame}

\begin{frame}{Base Problem}
	\begin{center}
		\Large\texttt{PUZ020-1.p}
	\end{center}

	{\large
	\begin{gather}
		a(\eta) \qquad \neg b(\eta) \notag\\
		\neg a(A) \lor \neg c\bigl(\phi(A)\bigr) \lor b(A) \notag\\
		c\bigl(\phi(\eta)\bigr) \lor b(\gamma)
		\qquad
		\neg b(\gamma) \lor c\bigl(\phi(\eta)\bigr) \notag
	\end{gather}
	}

	\small
	\begin{columns}[T,onlytextwidth]
		\column{.48\textwidth}
		$ a(A) $: $A$ is a person. \\
		$ b(A) $: $A$ is a knight. \\
		$ c(B) $: $B$ is a true statement. \\
		$ d(A) $: $A$ is a knave.

		\column{.48\textwidth}
		$ e(A,B) $: $A$ says statement $B$. \\
		$ \phi(A) $: statement $\phi$ made by $A$. \\
		$ \eta $: the husband. \\
		$ \gamma $: the wife.
	\end{columns}
\end{frame}
